SELECT item_name,
       CONCAT(first_name, ' ', last_name) AS borrower,
       contact_number,
       DATEDIFF(CURDATE(), due_date) AS days_late
FROM v_open_loans
WHERE is_overdue = 1
ORDER BY days_late DESC;

4. Attendance per Day

Sign-ins and how many were visitors.

daysign_insvisitorsstill_inside
2026-11-23724
2026-11-20400
 
SELECT DATE(a.time_in) AS day,
       COUNT(*) AS sign_ins,
       SUM(m.person_id IS NULL) AS visitors,
       SUM(a.time_out IS NULL) AS still_inside
FROM attendance a
LEFT JOIN member m ON m.person_id = a.person_id
GROUP BY DATE(a.time_in)
ORDER BY day DESC;

Why people come to the HQ.

SELECT p.name AS purpose,
       COUNT(*) AS visits,
       ROUND(AVG(TIMESTAMPDIFF(MINUTE, a.time_in, a.time_out))) AS avg_minutes
FROM attendance a
JOIN purpose p ON p.purpose_id = a.purpose_id
GROUP BY p.purpose_id, p.name
ORDER BY visits DESC;
-- AVG skips rows with no time_out, so people
-- still inside don’t distort the average

6. Most Borrowed Items

Approved and completed requests only.

nametimes_requestedtotal_quantity

SELECT i.name,
       COUNT(*) AS times_requested,
       SUM(rl.quantity) AS total_quantity
FROM request_line rl
JOIN item_request q ON q.request_id = rl.request_id
JOIN item i         ON i.item_id    = rl.item_id
WHERE q.status IN ('approved', 'completed')
GROUP BY i.item_id, i.name
ORDER BY total_quantity DESC, times_requested DESC
LIMIT 10;
