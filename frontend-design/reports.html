Attendance per day and purpose

Tables attendance, purpose

SELECT DATE(a.time_in) AS day, p.name AS purpose,
       COUNT(*) AS sign_ins
FROM attendance a
JOIN purpose p ON p.purpose_id = a.purpose_id
WHERE a.time_in >= CURDATE() - INTERVAL 1 DAY
GROUP BY day, p.name
ORDER BY day DESC, sign_ins DESC;

Most borrowed items

Tables request_line, item_request, item

SELECT i.name, i.item_type,
       SUM(rl.quantity) AS units_out
FROM request_line rl
JOIN item_request q ON q.request_id = rl.request_id
JOIN item i ON i.item_id = rl.item_id
WHERE q.status IN ('approved', 'completed')
GROUP BY i.item_id, i.name, i.item_type
ORDER BY units_out DESC
LIMIT 5;

Who is inside right now

Tables attendance, person, member

SELECT p.first_name, p.last_name,
       IF(m.person_id IS NULL, 'Visitor', 'Member') AS kind,
       a.time_in
FROM attendance a
JOIN person p      ON p.person_id = a.person_id
LEFT JOIN member m ON m.person_id = p.person_id
WHERE a.time_out IS NULL
ORDER BY a.time_in;
