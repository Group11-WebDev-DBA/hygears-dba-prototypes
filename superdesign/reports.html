4. In the HQ now

Open attendance cards: time_out IS NULL

SELECT p.first_name, p.last_name,
       pu.name AS purpose, a.time_in,
       IF(m.person_id IS NULL, 'Visitor', 'Member') AS kind
FROM attendance a
JOIN person p   ON p.person_id = a.person_id
JOIN purpose pu ON pu.purpose_id = a.purpose_id
LEFT JOIN member m ON m.person_id = p.person_id
WHERE a.time_out IS NULL
ORDER BY a.time_in;

5. Attendance per day and purpose

GROUP BY on attendance and purpose

SELECT DATE(a.time_in) AS day,
       pu.name AS purpose,
       COUNT(*) AS visits
FROM attendance a
JOIN purpose pu ON pu.purpose_id = a.purpose_id
GROUP BY DATE(a.time_in), pu.name
ORDER BY day DESC, visits DESC;

6. Most borrowed items

Approved and completed request lines

SELECT i.name, i.item_type,
       SUM(rl.quantity) AS times_out
FROM request_line rl
JOIN item_request q ON q.request_id = rl.request_id
JOIN item i ON i.item_id = rl.item_id
WHERE q.status IN ('approved', 'completed')
GROUP BY i.item_id, i.name, i.item_type
ORDER BY times_out DESC
LIMIT 5;
