4. Attendance per day

table attendance
SELECT DATE(time_in)             AS day,
       COUNT(*)                  AS sign_ins,
       COUNT(DISTINCT person_id) AS people
FROM   attendance
WHERE  time_in >= CURDATE() - INTERVAL 7 DAY
GROUP  BY DATE(time_in)
ORDER  BY day DESC;

6. Most borrowed items

request_line JOIN item_request JOIN item
SELECT i.name, i.item_type,
       SUM(rl.quantity)             AS total_qty,
       COUNT(DISTINCT rl.request_id) AS requests
FROM   request_line rl
JOIN   item_request q ON q.request_id = rl.request_id
JOIN   item i         ON i.item_id    = rl.item_id
WHERE  q.status IN ('approved', 'completed')
GROUP  BY i.item_id, i.name, i.item_type
ORDER  BY requests DESC, total_qty DESC
LIMIT  5;
