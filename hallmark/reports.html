4. Attendance per day

Visits, split into members and visitors

visit_datevisitsmembersvisitors
2026-11-23642
2026-11-21431
 
SELECT DATE(a.time_in) AS visit_date,
       COUNT(*) AS visits,
       COUNT(m.person_id) AS members,
       COUNT(*) - COUNT(m.person_id) AS visitors
FROM attendance a
LEFT JOIN member m ON m.person_id = a.person_id
GROUP BY DATE(a.time_in)
ORDER BY visit_date DESC;

SELECT p.name AS purpose, COUNT(*) AS visits
FROM attendance a
JOIN purpose p ON p.purpose_id = a.purpose_id
WHERE a.time_in >= DATE_FORMAT(CURDATE(), '%Y-%m-01')
GROUP BY p.purpose_id, p.name
ORDER BY visits DESC;

SELECT i.name, i.item_type,
       COUNT(*) AS times_requested,
       SUM(rl.quantity) AS total_quantity
FROM request_line rl
JOIN item_request q ON q.request_id = rl.request_id
JOIN item i ON i.item_id = rl.item_id
WHERE q.status IN ('approved', 'completed')
GROUP BY i.item_id, i.name, i.item_type
ORDER BY times_requested DESC
LIMIT 5;
