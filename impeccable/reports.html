Who is inside now

attendance rows with no time_out yet

SELECT p.first_name, p.last_name,
       pu.name AS purpose,
       a.time_in,
       TIMEDIFF(NOW(), a.time_in) AS inside_for
FROM attendance a
JOIN person  p  ON p.person_id  = a.person_id
JOIN purpose pu ON pu.purpose_id = a.purpose_id
WHERE a.time_out IS NULL
ORDER BY a.time_in;

Overdue tools

same view, only rows past due_date, with a contact number to call

SELECT first_name, last_name, contact_number,
       item_name, due_date,
       DATEDIFF(CURDATE(), due_date) AS days_late
FROM v_open_loans
WHERE is_overdue = 1
ORDER BY days_late DESC;

Attendance per day and purpose

counts of sign-ins grouped by date and purpose

SELECT DATE(a.time_in) AS day,
       pu.name AS purpose,
       COUNT(*) AS visits
FROM attendance a
JOIN purpose pu ON pu.purpose_id = a.purpose_id
GROUP BY DATE(a.time_in), pu.name
ORDER BY day DESC, visits DESC;

Most borrowed items

total quantity on approved and completed requests

SELECT i.name, i.item_type,
       COUNT(DISTINCT rl.request_id) AS requests,
       SUM(rl.quantity) AS total_qty
FROM request_line rl
JOIN item_request q ON q.request_id = rl.request_id
JOIN item i         ON i.item_id    = rl.item_id
WHERE q.status IN ('approved', 'completed')
GROUP BY i.item_id, i.name, i.item_type
ORDER BY total_qty DESC
LIMIT 5;
